% A worked example of the cktimg package.
%
% With shell escape (pdflatex -shell-escape figure.tex) the figure is drawn
% during the run. Without it (Overleaf, most CI), draw it first:
%
%   cktimg-tex cs.cir cs.cktimg.tex
%
% and compile normally; \cktimg then inputs what you generated.
\documentclass{article}
\usepackage{cktimg}

\begin{document}
\section*{A common-source stage}

\begin{figure}[h]
  \centering
  \cktimg{cs.cir}
  \caption{The stage, drawn from its netlist. The supply and input sources
    stand apart on the left; the circuit uses their terminal symbols.}
\end{figure}

With your own settings or symbols:
\verb|\cktimg[--config lint.zon --symbols mine.zon]{cs.cir}|.
\end{document}
